### 1. 物理感知层：多级数字滤波与时分复用测距 (`ultrasonic.c`)
报告指出，原始超声波数据充满尖峰噪声，直接输入PID会导致电机剧烈抽搐 。您现有的简单均值滤波需要升级为中值滤波（消除孤立脉冲）+ 指数加权移动平均（EMA，平滑台阶效应）的级联架构 。此外，为了防止三个超声波相互干扰，需要引入时分复用触发 。
将您的 `ultrasonic.c` 中滤波与读取部分改良为：

```c
#include "ultrasonic.h"

#define FILTER_SIZE 5
// 平滑系数：前方响应要快，侧方求平滑防颤振
#define ALPHA_FRONT 0.7f
#define ALPHA_SIDE  0.35f

// 冒泡排序用于中值提取
static void BubbleSort(float *arr, int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                float temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

// 级联滤波器：滑动中值 + EMA低通滤波
float Filter_Distance(float new_raw_val, float *ring_buf, uint8_t *buf_idx, float *last_ema, float alpha) {
    // 1. 滑动窗口更新
    ring_buf[*buf_idx] = (new_raw_val > 400.0f) ? 400.0f : new_raw_val;
    *buf_idx = (*buf_idx + 1) % FILTER_SIZE;
    
    // 2. 复制并排序取中值 [cite: 28]
    float sort_buf[FILTER_SIZE];
    for (int i = 0; i < FILTER_SIZE; i++) sort_buf[i] = ring_buf[i];
    BubbleSort(sort_buf, FILTER_SIZE);
    float median_val = sort_buf[FILTER_SIZE / 2];
    
    // 3. EMA滤波 [cite: 29, 30]
    float ema_val = alpha * median_val + (1.0f - alpha) * (*last_ema);
    *last_ema = ema_val;
    
    return ema_val;
}
```

```c
typedef enum {
    NAV_STATE_IDLE,
    NAV_STATE_CRUISE,       // 正常直行，PID自主跟随
    NAV_STATE_SCAN,         // 道路跳变，进行死区延时屏蔽防虚假节点

    NAV_STATE_TURN          // 混合闭环转向

static PID_t pid_wall;
static uint32_t state_start_time = 0;

#define MIN_PWM 0
#define WALL_TARGET_DIST 10.0f // 单侧巡墙目标距离

void App_Init(void) {
    // 初始化PID：需根据实车调试 Kp, Ki, Kd
    PID_Init(&pid_wall, 2.5f, 0.1f, 1.0f);
}

// 约束函数
int clamp(int val, int min, int max) {
    return (val < min) ? min : ((val > max) ? max : val);
}

// 核心更新循环：需要传入左、前、右三个滤波后的超声波数据
void App_Update(float D_left, float D_front, float D_right) {
    switch(state) {
        case NAV_STATE_CRUISE: {
            float error = 0;
            // 模态A：双侧有墙走廊居中 [cite: 40, 41]
            if (D_left < 20.0f && D_right < 20.0f) {

            else if (D_left < 20.0f) {
                error = D_left - WALL_TARGET_DIST;
            } else if (D_right < 20.0f) {
                error = WALL_TARGET_DIST - D_right; // 镜像翻转 [cite: 47]

            // ... (微调代码)
            state = NAV_STATE_TURN; // 对齐后准备转向
            break;
        }

        case NAV_STATE_TURN: {

            uint32_t t_elapsed = HAL_GetTick() - state_start_time;

            }
            break;
        }
    }
}
```

### 3. 高层决策：时序阈值与图论拓扑 (`maze.c`)
您原有的 `maze.c` 使用的是极其单薄的单次大小比较。根据论文的“单侧逻辑状态判定机制”，迷宫由时间序列测距抽象而来 。应当使用多次采样确认，并映射出物理拓扑节点（左侧T型、死胡同等）。

```c
#include "maze.h"

#define THRESHOLD_WALL 25.0f // 论文推荐的侧向阈值 [cite: 90]
#define THRESHOLD_FRONT 10.0f // 前方制动阈值 [cite: 90]

// 映射基于论文中的"传感器状态向量图表" [cite: 92, 97, 98, 101]
Action_t Maze_Decide_Topology(float L, float F, float R) {
    uint8_t left_open  = (L > THRESHOLD_WALL);
    uint8_t front_open = (F > THRESHOLD_FRONT);
    uint8_t right_open = (R > THRESHOLD_WALL);

    // [OPEN, OPEN, OPEN] 十字交叉口 [cite: 108]
    if (left_open && front_open && right_open) return TURN_RIGHT; // 默认右手定则优先级1 [cite: 119]
    
    // 右侧T型岔路 [cite: 108]
    if (!left_open && front_open && right_open) return TURN_RIGHT;
    
    // 左侧T型岔路 [cite: 101]
    if (left_open && front_open && !right_open) return MOVE_FORWARD; // 右侧无路，优先直行 [cite: 120]
    
    // 正交T型墙壁 [cite: 108]
    if (left_open && !front_open && right_open) return TURN_RIGHT;
    
    // 左转死角 [cite: 99]
    if (left_open && !front_open && !right_open) return TURN_LEFT;
    
    // 右转死角 [cite: 100]
    if (!left_open && !front_open && right_open) return TURN_RIGHT;
    
    // 死胡同端点 [cite: 98, 122]
    if (!left_open && !front_open && !right_open) return TURN_BACK; 

    // 标准直行通道 [cite: 97]
    return MOVE_FORWARD;
}
```
